// Haptik-Test: ESP32 + DRV2605L auf dem Breadboard
// Verdrahtung: 3V3 -> VIN, GND -> GND, GPIO21 -> SDA, GPIO22 -> SCL
// Motor an die Motor-Klemmen (+/-) des DRV2605L-Boards.
// Taster: GPIO19 -> Taster -> GND. Jeder Druck spielt den nächsten Effekt.

#include <Arduino.h>
#include <Wire.h>
#include <Adafruit_DRV2605.h>

// Motortyp: true = LRA (Linearaktor), false = ERM (Rundmotor mit Unwucht)
#define MOTOR_IST_LRA false

#define PIN_SDA 21
#define PIN_SCL 22
#define PIN_TASTER 19

Adafruit_DRV2605 drv;

struct Effekt {
  uint8_t nummer;
  const char* name;
};

const Effekt effekte[] = {
  {1,   "Starker Klick"},
  {4,   "Scharfer Klick"},
  {10,  "Doppelklick"},
  {14,  "Starkes Brummen"},
  {27,  "Kurzer Doppelklick, leicht"},
  {47,  "Brummen (Buzz)"},
  {52,  "Pulsierend, stark"},
  {118, "Langer Summton"},
};
const uint8_t ANZAHL = sizeof(effekte) / sizeof(effekte[0]);
uint8_t aktuellerEffekt = 0;

void i2cScan() {
  Serial.println("I2C-Scan ...");
  uint8_t gefunden = 0;
  for (uint8_t a = 1; a < 127; a++) {
    Wire.beginTransmission(a);
    if (Wire.endTransmission() == 0) {
      Serial.printf("  Gerät bei 0x%02X\n", a);
      gefunden++;
    }
  }
  if (gefunden == 0) {
    Serial.println("  Nichts gefunden. SDA/SCL, 3V3 und GND prüfen.");
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_TASTER, INPUT_PULLUP);
  delay(500);
  Serial.println("\n=== Haptik-Test DRV2605L ===");

  Wire.begin(PIN_SDA, PIN_SCL);
  i2cScan();

  if (!drv.begin(&Wire)) {
    Serial.println("DRV2605L nicht gefunden (erwartet 0x5A). Verkabelung prüfen!");
    while (true) delay(1000);
  }
  Serial.println("DRV2605L gefunden.");

  if (MOTOR_IST_LRA) {
    drv.useLRA();
    drv.selectLibrary(6);   // Bibliothek 6 = LRA
  } else {
    drv.useERM();
    drv.selectLibrary(1);   // Bibliothek 1 = ERM
  }
  drv.setMode(DRV2605_MODE_INTTRIG);
}

void loop() {
  static bool warGedrueckt = false;
  bool gedrueckt = digitalRead(PIN_TASTER) == LOW;

  if (gedrueckt && !warGedrueckt) {
    const Effekt& e = effekte[aktuellerEffekt];
    Serial.printf("Effekt %3u: %s\n", e.nummer, e.name);
    drv.setWaveform(0, e.nummer);
    drv.setWaveform(1, 0);   // Ende der Sequenz
    drv.go();
    aktuellerEffekt = (aktuellerEffekt + 1) % ANZAHL;
  }
  warGedrueckt = gedrueckt;

  delay(30);   // Entprellen
}
